// Estrutura real de PELADAS. CONFIG não é lida nem alterada por este fluxo.
const PELADA_FIELDS=['pelada_id','nome','local','dia_semana','hora_inicio','hora_fim','mensalidade','diaria_sem_colete','diaria_com_colete','colete_avulso','vencimento_dia','convite_horas','owner_id','situacao'];
function peladaValid(p) {
 return p && ['nome','local','cidade'].every(k=>typeof p[k]==='string' && p[k].trim().length>0 && p[k].length<=150)
 && Number.isInteger(p.dia_semana) && p.dia_semana>=0 && p.dia_semana<=6
 && ['horario_inicio','horario_fim'].every(k=>typeof p[k]==='string' && /^([01]\d|2[0-3]):[0-5]\d$/.test(p[k])) && p.horario_fim>p.horario_inicio
 && ['mensalidade','diaria_sem_colete','diaria_com_colete','colete_avulso'].every(k=>typeof p[k]==='number' && Number.isFinite(p[k]) && p[k]>=0 && p[k]<=100000 && Math.abs(p[k]*100-Math.round(p[k]*100))<0.000001)
 && Number.isInteger(p.vencimento_mensal) && p.vencimento_mensal>=1 && p.vencimento_mensal<=28
 && Number.isInteger(p.validade_convite_horas) && p.validade_convite_horas>=1 && p.validade_convite_horas<=720;
}
function readTable(snapshot,index,name,fields) {
 const t=snapshot.valueRanges?.[index];const values=t?.values;
 if(!t || !String(t.range??'').replace(/'/g,'').startsWith(name+'!') || !Array.isArray(values) || fields.some((k,i)=>values[0]?.[i]!==k) || values.length>5001) throw new Error('schema');
 return values.slice(1).filter(r=>r.some(v=>String(v??'').trim())).map(r=>Object.fromEntries(fields.map((k,i)=>[k,r[i]??''])));
}
function localComCidade(p) {return p.local.toLowerCase().includes(p.cidade.trim().toLowerCase())?p.local.trim():p.local.trim()+' — '+p.cidade.trim();}
function peladaValues(p,id,owner) {
 return {pelada_id:id,nome:p.nome.trim(),local:localComCidade(p),dia_semana:p.dia_semana,hora_inicio:p.horario_inicio,hora_fim:p.horario_fim,mensalidade:p.mensalidade,diaria_sem_colete:p.diaria_sem_colete,diaria_com_colete:p.diaria_com_colete,colete_avulso:p.colete_avulso,vencimento_dia:p.vencimento_mensal,convite_horas:p.validade_convite_horas,owner_id:owner,situacao:'ativo'};
}
function readPelada(row) {
 const p={id:row.pelada_id,nome:row.nome,local:row.local,cidade:'',dia_semana:Number(row.dia_semana),horario_inicio:row.hora_inicio,horario_fim:row.hora_fim,mensalidade:Number(row.mensalidade),diaria_sem_colete:Number(row.diaria_sem_colete),diaria_com_colete:Number(row.diaria_com_colete),colete_avulso:Number(row.colete_avulso),vencimento_mensal:Number(row.vencimento_dia),validade_convite_horas:Number(row.convite_horas)};
 if(!peladaValid({...p,cidade:'cidade no local'}) || row.situacao!=='ativo')throw new Error('pelada_invalid');
 return p;
}
function selectedPelada(request) {
 const value=request.query?.pelada_id ?? request.body?.pelada_id ?? '';
 return typeof value==='string' && value.length<=150 && (!value || /^[A-Za-z0-9_-]+$/.test(value)) ? value : null;
}
function prepareSession(request,config) {
 const p=prepare('sessao',request,config),selected=selectedPelada(request);
 return selected===null?{...reply(400,'PELADA_INVALIDA','Seleção inválida.'),canRun:false}:{...p,selectedPelada:selected};
}
function prepareBootstrap(request,config,authenticated=false) {
 const body=request.body;
 const headers=Object.fromEntries(Object.entries(request.headers??{}).map(([k,v])=>[k.toLowerCase(),v]));
 const key=String(headers['idempotency-key']??'').toLowerCase();
 const bad=()=>({...reply(400,'DADOS_INVALIDOS','Confira os dados da pelada e da operação.'),canRun:false});
 if(!body || typeof body!=='object' || Array.isArray(body) || !peladaValid(body.pelada) || !/^[a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{12}$/.test(key))return bad();
 if(!Number.isInteger(config.gid_peladas)||config.gid_peladas<0)return {...reply(503,'CONFIGURACAO_PENDENTE','Confira a configuração da planilha.'),canRun:false};
 if(authenticated)return {...prepare('sessao',request,config),authenticated:true,pelada:body.pelada,requestId:key};
 const o=body.organizador;
 if(!o || !['','Mensalista','Diarista'].includes(o.categoria??'') || typeof o.apelido!=='string' || !o.apelido.trim() || o.apelido.length>40)return bad();
 const p=prepare('cadastro_owner',{...request,body:o},config);
 return {...p,profile:p.profile?{...p.profile,apelido:o.apelido.trim(),categoria:o.categoria??''}:null,pelada:body.pelada,requestId:key};
}
function planBootstrap(prepared,snapshot,hashes,nowMs) {
 try {
  if(!prepared.canRun)return reply(400,'DADOS_INVALIDOS','Operação inválida.');
  const players=readTable(snapshot,0,'JOGADORES',PLAYER_FIELDS);
  const clubs=readTable(snapshot,3,'PELADAS',PELADA_FIELDS);
  if(!Array.isArray(snapshot.namedRanges) || new Set(clubs.map(c=>c.pelada_id)).size!==clubs.length)throw new Error('metadata');
  const byId=id=>players.filter(r=>r.participante_id===id);
  let owner;
  let authRequests=[];
  if(prepared.authenticated) {
   const access=plan(prepared,snapshot,hashes,nowMs);
   if(access.status!==200 || !access.response.autenticado)return access;
   authRequests=access.requests;
   const found=byId(access.response.participante_id);
   if(found.length!==1)throw new Error('identity');
   owner=found[0];
   if(String(owner.papel_administrativo).toUpperCase()!=='OWNER')return reply(403,'ACESSO_RESTRITO','Somente o OWNER pode criar peladas.');
  } else {
   const found=players.filter(r=>digits(r.cpf)===prepared.cpf);
   if(found.length>1)throw new Error('identity');
   owner=found[0];
  }
  const claimId='rc5_club_req_'+prepared.requestId.replace(/-/g,'_');
  const markers=snapshot.namedRanges.filter(r=>r.namedRangeId===claimId);
  // Compatibilidade de repetição com o bootstrap antigo, sem bloqueio global.
  const legacy=snapshot.namedRanges.find(r=>r.namedRangeId==='rc5_owner_bootstrap_v1' && r.name==='RC5_OWNER_'+prepared.requestId.replace(/-/g,'_'));
  if(markers.length>1)throw new Error('marker');
  const marker=markers[0];
  if(marker || legacy) {
   const club=marker?clubs.find(c=>'RC5_CLUB_'+c.pelada_id===marker.name):clubs.find(c=>owner && c.pelada_id===owner.pelada_id && c.owner_id===owner.participante_id);
   const profileMatches=prepared.authenticated || owner && owner.senha_hash===hashes.passwordHash && ['nome','email','telefone','apelido','categoria'].every(k=>owner[k]===prepared.profile[k]);
   const same=owner && String(owner.papel_administrativo).toUpperCase()==='OWNER' && owner.situacao==='ativo' && club && club.owner_id===owner.participante_id && profileMatches && Object.entries(peladaValues(prepared.pelada,club.pelada_id,owner.participante_id)).every(([k,v])=>String(club[k])===String(v));
   return same?reply(200,'PELADA_CRIADA','Pelada já criada.',{participante_id:owner.participante_id,pelada_id:club.pelada_id}):reply(409,'REQUISICAO_DIVERGENTE','Operação já utilizada. Confira o cadastro antes de repetir.');
  }
  if(!prepared.authenticated && owner)return reply(409,'CADASTRO_EXISTENTE','Faça login para criar outra pelada.');
  if(!/^[a-f0-9]{64}$/i.test(hashes.candidate))throw new Error('crypto');
  const peladaId='PELADA_'+hashes.candidate;
  if(clubs.some(c=>c.pelada_id===peladaId))throw new Error('collision');
  const ownerId=owner?.participante_id || 'OWNER_'+hashes.candidate;
  if(!owner && byId(ownerId).length)throw new Error('collision');
  const requests=[...authRequests];
  const cell=v=>({userEnteredValue:typeof v==='number'?{numberValue:v}:{stringValue:String(v??'')}});
  const append=(gid,fields,row)=>requests.push({appendCells:{sheetId:gid,rows:[{values:fields.map(k=>cell(row[k]))}],fields:'userEnteredValue'}});
  const claim=(id,name,gid)=>requests.push({addNamedRange:{namedRange:{namedRangeId:id,name,range:{sheetId:gid,startRowIndex:0,endRowIndex:1,startColumnIndex:0,endColumnIndex:1}}}});
  claim(claimId,'RC5_CLUB_'+peladaId,prepared.config.gid_peladas);
  if(!owner) {
   if(!/^[a-f0-9]{64}$/i.test(hashes.passwordHash)||!/^[a-f0-9]{64}$/i.test(hashes.cpfHash))throw new Error('crypto');
   const cpfClaim='rc5_owner_cpf_'+hashes.cpfHash;
   if(snapshot.namedRanges.some(r=>r.namedRangeId===cpfClaim))return reply(409,'CADASTRO_EXISTENTE','Confira o cadastro existente antes de repetir.');
   claim(cpfClaim,'RC5_OWNER_CPF_'+hashes.cpfHash,prepared.config.gid_jogadores);
   const now=new Date(nowMs).toISOString();
   append(prepared.config.gid_jogadores,PLAYER_FIELDS,{...prepared.profile,participante_id:ownerId,pelada_id:peladaId,cpf:prepared.cpf,papel_administrativo:'OWNER',situacao:'ativo',senha_hash:hashes.passwordHash,criado_em:now,atualizado_em:now});
  }
  append(prepared.config.gid_peladas,PELADA_FIELDS,peladaValues(prepared.pelada,peladaId,ownerId));
  return {...reply(201,'PELADA_CRIADA','Pelada criada.',{participante_id:ownerId,pelada_id:peladaId}),requests};
 }catch{return reply(503,'DADOS_INCOMPATIVEIS','Não foi possível confirmar a criação. Confira JOGADORES, PELADAS e os marcadores antes de repetir.');}
}
function planSession(prepared,snapshot,hashes,nowMs) {
 const result=plan(prepared,snapshot,hashes,nowMs);
 if(result.status!==200 || !result.response.autenticado)return result;
 try {
  const players=readTable(snapshot,0,'JOGADORES',PLAYER_FIELDS);
  const owners=players.filter(r=>r.participante_id===result.response.participante_id);
  if(owners.length!==1)throw new Error('owner');
  const rows=readTable(snapshot,3,'PELADAS',PELADA_FIELDS);
  if(new Set(rows.map(c=>c.pelada_id)).size!==rows.length)throw new Error('duplicate_club');
  const user=owners[0];
  const role=String(user.papel_administrativo).trim().toUpperCase();
  if(!['OWNER','ADMIN',''].includes(role) || (!role && !['MENSALISTA','DIARISTA'].includes(String(user.categoria).toUpperCase())))return reply(403,'PERFIL_INVALIDO','Perfil não autorizado.');
  const clubs=rows.filter(r=>r.situacao==='ativo' && (role==='OWNER'?r.owner_id===user.participante_id:r.pelada_id===user.pelada_id));
  const selected=prepared.selectedPelada || clubs.find(c=>c.pelada_id===owners[0].pelada_id)?.pelada_id || clubs[0]?.pelada_id;
  const club=clubs.find(c=>c.pelada_id===selected);
  if(prepared.selectedPelada && !club)return reply(403,'PELADA_NAO_AUTORIZADA','Você não administra essa pelada.');
  const p=club?readPelada(club):null;
  if(!p && role!=='OWNER')return reply(403,'PELADA_NAO_AUTORIZADA','Você não possui vínculo ativo com uma pelada.');
  const admin=role==='OWNER'||role==='ADMIN';
  return {...result,response:{...result.response,papel_administrativo:role,versao_acesso:2,permissoes:{criar_pelada:role==='OWNER',administrar:admin},meu_cadastro:publicProfile(user),peladas:clubs.map(readPelada),pelada:p,...(admin?{total_jogadores:p?players.filter(r=>r.pelada_id===p.id && ['MENSALISTA','DIARISTA'].includes(String(r.categoria).toUpperCase()) && r.situacao==='ativo').length:0}:{})}};
 }catch{return reply(503,'PELADA_INDISPONIVEL','Não foi possível carregar as peladas do organizador.');}
}
// Somente o titular autenticado. Nenhum campo de permissão, senha ou financeiro é editável.
function publicProfile(user) {
 return {nome:user.nome,cpf:user.cpf,apelido:user.apelido,telefone:user.telefone,email:user.email,contato_editavel:!String(user.asaas_customer_id).trim()};
}
function prepareProfile(request,config) {
 const p=prepareSession(request,config);
 if(!p.canRun)return p;
 const body=request.body;
 if(!body || typeof body!=='object' || Array.isArray(body) || Object.keys(body).some(k=>!['apelido','telefone','email'].includes(k)) || typeof body.apelido!=='string' || !body.apelido.trim() || body.apelido.length>40 || typeof body.telefone!=='string' || !/^\d{10,13}$/.test(digits(body.telefone)) || typeof body.email!=='string' || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(body.email) || body.email.length>254)return {...reply(400,'DADOS_INVALIDOS','Confira apelido, telefone e e-mail. Apenas esses campos são aceitos.'),canRun:false};
 return {...p,profileUpdate:{apelido:body.apelido.trim(),telefone:digits(body.telefone),email:body.email.trim().toLowerCase()}};
}
function planProfile(prepared,snapshot,hashes,nowMs) {
 if(!prepared.canRun)return reply(400,'DADOS_INVALIDOS','Confira os dados pessoais.');
 const access=plan(prepared,snapshot,hashes,nowMs);
 if(access.status!==200 || !access.response.autenticado)return access;
 try {
  const players=readTable(snapshot,0,'JOGADORES',PLAYER_FIELDS);
  const found=players.filter(r=>r.participante_id===access.response.participante_id);
  if(found.length!==1)throw new Error('identity');
  const user=found[0],update=prepared.profileUpdate;
  if(!update)throw new Error('input');
  const linked=Boolean(String(user.asaas_customer_id).trim());
  if(linked && (update.telefone!==digits(user.telefone) || update.email!==String(user.email).toLowerCase().trim()))return reply(409,'CONTATO_VINCULADO_ASAAS','Telefone e e-mail vinculados ao Asaas estão disponíveis somente para consulta nesta etapa.');
  const raw=snapshot.valueRanges[0].values;
  const row=raw.findIndex((r,i)=>i>0 && r[0]===user.participante_id);
  if(row<1)throw new Error('row');
  const requests=[...access.requests];
  const fields={...(linked?{apelido:update.apelido}:update),atualizado_em:new Date(nowMs).toISOString()};
  for(const [key,value] of Object.entries(fields))requests.push({updateCells:{start:{sheetId:prepared.config.gid_jogadores,rowIndex:row,columnIndex:PLAYER_FIELDS.indexOf(key)},rows:[{values:[{userEnteredValue:{stringValue:value}}]}],fields:'userEnteredValue'}});
  return {...reply(200,'CADASTRO_ATUALIZADO','Seus dados foram atualizados.',{meu_cadastro:publicProfile({...user,...fields})}),requests};
 }catch{return reply(503,'CADASTRO_INDISPONIVEL','Não foi possível confirmar a atualização. Consulte seu cadastro antes de repetir.');}
}
